from flask import Flask, render_template_string, request
from flask_sqlalchemy import SQLAlchemy
from datetime import datetime

app = Flask(__name__)
app.config['SQLALCHEMY_DATABASE_URI'] = 'sqlite:///hospital.db'
db = SQLAlchemy(app)

class State(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    name = db.Column(db.String(50))

class Hospital(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    name = db.Column(db.String(100))
    type = db.Column(db.String(20)) # Govt ya Private
    state_id = db.Column(db.Integer)

class OPD(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    opd_number = db.Column(db.String(10))
    department = db.Column(db.String(50))
    hospital_id = db.Column(db.Integer)

class Token(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    token_number = db.Column(db.String(10))
    token_type = db.Column(db.String(20)) # Normal/Emergency
    opd_id = db.Column(db.Integer)
    status = db.Column(db.String(20), default='Waiting')
    time = db.Column(db.DateTime, default=datetime.now)

@app.route('/')
def home():
    states = State.query.all()
    return render_template_string('''
    <h1>🏥 Digital Hospital Token - Maharashtra</h1>
    <form action="/hospitals">
        <h3>State chuno:</h3>
        <select name="state_id">
        {% for s in states %}
            <option value="{{s.id}}">{{s.name}}</option>
        {% endfor %}
        </select>
        <button type="submit">Hospital dekho</button>
    </form>
    ''', states=states)

@app.route('/hospitals')
def hospitals():
    state_id = request.args.get('state_id')
    hospitals = Hospital.query.filter_by(state_id=state_id).all()
    return render_template_string('''
    <h2>Maharashtra ke Hospital</h2>
    {% for h in hospitals %}
        <div style="border:1px solid #ccc; padding:10px; margin:10px;">
        <b>{{h.name}}</b> <span style="color:green">[{{h.type}}]</span><br>
        <a href="/opd/{{h.id}}">Token lo</a>
        </div>
    {% endfor %}
    ''', hospitals=hospitals)

if __name__ == '__main__':
    with app.app_context():
        db.create_all()
        if State.query.count() == 0:
            mh = State(name='Maharashtra')
            db.session.add(mh)

            # Govt Hospitals
            h1 = Hospital(name='JJ Hospital, Mumbai', type='Govt', state_id=mh.id)

            h3 = Hospital(name='Mayo Hospital, Nagpur', type='Govt', state_id=mh.id)
            
            # Private Hospitals  
            h4 = Hospital(name='Lilavati Hospital, Mumbai', type='Private', state_id=mh.id)
            h5 = Hospital(name='Jehangir Hospital, Pune', type='Private', state_id=mh.id)
            
            db.session.add_all([h1,h2,h3,h4,h5])

    app.run(debug=True)
